package consumer

import (
    "bytes"
    "context"
    "net/http/httptest"
    "strings"
    "testing"

    "github.com/Tend-Stack/TendUI/v2026"
    "github.com/Tend-Stack/TendUI/v2026/components"
    "github.com/Tend-Stack/TendUI/v2026/render"
)

func TestPublishedModule(t *testing.T) {
    if tendui.Version != "2026.1.0" { t.Fatalf("unexpected version %q", tendui.Version) }
    var output bytes.Buffer
    button := components.Button(components.ButtonProps{Text: "Published <Go>", Primary: true})
    page := components.Layout(components.LayoutProps{Title: "Consumer", Language: "en", Theme: components.ThemeDark}, button)
    if err := page.Render(context.Background(), &output); err != nil { t.Fatal(err) }
    for _, want := range []string{"Published &lt;Go&gt;", `data-theme="dark"`, tendui.StylesheetURL} {
        if !strings.Contains(output.String(), want) { t.Fatalf("missing %q", want) }
    }
    if strings.Contains(output.String(), "<script") { t.Fatal("unexpected bundled script") }
    css := httptest.NewRecorder()
    tendui.AssetHandler().ServeHTTP(css, httptest.NewRequest("GET", tendui.StylesheetURL, nil))
    if css.Code != 200 || !strings.Contains(css.Body.String(), "--tendui-primary") { t.Fatal("embedded asset not available") }
    request := httptest.NewRequest("GET", "/", nil)
    request.Header.Set("HX-Request", "true")
    fragment := httptest.NewRecorder()
    if err := render.HTML(fragment, request, 200, page, button); err != nil { t.Fatal(err) }
    if strings.Contains(fragment.Body.String(), "<html") || !strings.Contains(fragment.Header().Get("Vary"), "HX-Request") { t.Fatal("fragment contract failed") }
}
